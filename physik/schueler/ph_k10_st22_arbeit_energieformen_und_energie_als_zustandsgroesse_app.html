/*
  Klasse 10 – Arbeit, Energieformen und Energie als Zustandsgröße
  Funktion: Kraftsensor und Wegsensor
  Sensor/Aktor: Loadcell/HX711, Ultraschall
  Messgröße/Rohwert: Kraft in N; Weg in m; Arbeit in J

  WICHTIG:
  - Nur Schutzkleinspannung verwenden.
  - Aufbau vor dem Anschließen des USB-Kabels kontrollieren.
  - Externe Lasten niemals direkt aus einem Arduino-Pin versorgen.
*/

const byte TRIG_PIN = 8;
const byte ECHO_PIN = 9;

float abstandCm() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  unsigned long dauer = pulseIn(ECHO_PIN, HIGH, 30000UL);
  if (dauer == 0) return -1.0;
  return dauer * 0.0343f / 2.0f;
}

void setup() {
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  Serial.begin(9600);
  Serial.println(F("Zeit_ms;Abstand_cm"));
}

void loop() {
  float d = abstandCm();
  if (d > 0) {
    Serial.print(millis());
    Serial.print(';');
    Serial.println(d, 1);
  }
  delay(100);
}
Diagnosehilfe
  • Kein Port: USB-Kabel, Board und Port prüfen.
  • Keine Werte: GND, Versorgung, Pin und Baudrate 9600 prüfen.
  • Nur 0/Maximalwert: Sensoranschluss und Spannungsteiler prüfen.
  • Starke Schwankungen: Kontakte sichern, Mittelwerte bilden und Messbedingungen konstant halten.
